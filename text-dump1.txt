# Script to insert data from courses.csv and students.csv into students database

PSQL="psql -X --username=freecodecamp --dbname=students --no-align --tuples-only -c"
echo $($PSQL "TRUNCATE students, majors, courses, majors_courses")

cat courses_test.csv | while IFS="," read MAJOR COURSE
do
  if [[ $MAJOR != "major" ]]
  then
    # get major_id
    MAJOR_ID=$($PSQL "SELECT major_id FROM majors WHERE major='$MAJOR'")

    # if not found
    if [[ -z $MAJOR_ID ]]
    then
      # insert major
      INSERT_MAJOR_RESULT=$($PSQL "INSERT INTO majors(major) VALUES('$MAJOR')")
      if [[ $INSERT_MAJOR_RESULT == "INSERT 0 1" ]]
      then
        echo Inserted into majors, $MAJOR
      fi

      # get new major_id
      MAJOR_ID=$($PSQL "SELECT major_id FROM majors WHERE major='$MAJOR'")
    fi

    # get course_id
    COURSE_ID=$($PSQL "SELECT course_id FROM courses WHERE course='$COURSE'")

    # if not found
    if [[ -z $COURSE_ID ]]
    then
      # insert course
      INSERT_COURSE_RESULT=$($PSQL "INSERT INTO courses(course) VALUES('$COURSE')")
      if [[ $INSERT_COURSE_RESULT == "INSERT 0 1" ]]
      then
        echo Inserted into courses, $COURSE
      fi

      # get new course_id
      COURSE_ID=$($PSQL "SELECT course_id FROM courses WHERE course='$COURSE'")
    fi

    # insert into majors_courses
    INSERT_MAJORS_COURSES_RESULT=$($PSQL "INSERT INTO majors_courses(major_id, course_id) VALUES($MAJOR_ID, $COURSE_ID)")
    if [[ $INSERT_MAJORS_COURSES_RESULT == "INSERT 0 1" ]]
    then
      echo Inserted into majors_courses, $MAJOR : $COURSE
    fi
  fi
done

function permAlone(str) {
  // Convert string to an array to swap characters easily
  let arr = str.split("");
  let count = 0;

  // Function to swap elements in the array
  function swap(idx1, idx2) {
    let temp = arr[idx1];
    arr[idx1] = arr[idx2];
    arr[idx2] = temp;
  }

  // Heap's Algorithm: generates all permutations
  function generate(n) {
    // Base Case: check for repeated consecutive letters
    if (n === 1) {
      // RegEx checks for any character followed immediately by itself
      if (!/(.)\1/.test(arr.join(""))) {
        count++;
      }
      return;
    }

    for (let i = 0; i < n; i++) {
      generate(n - 1);

      // If n is even, swap the i-th and last element
      // If n is odd, swap the first and last element
      if (n % 2 === 0) {
        swap(i, n - 1);
      } else {
        swap(0, n - 1);
      }
    }
  }

  generate(arr.length);
  return count;
}

function countNoRepeatsIterative(str) {
  const arr = str.split("");
  const n = arr.length;
  const c = new Array(n).fill(0); // Tracking array for Heap's Algorithm
  const regex = /(.)\1+/;
  let count = 0;

  // Helper to check for repeats
  const hasNoRepeats = (a) => !regex.test(a.join(""));

  // Check the initial permutation
  if (hasNoRepeats(arr)) count++;

  let i = 1;
  while (i < n) {
    if (c[i] < i) {
      // Swap elements based on whether the index is even or odd
      const swapIndex = i % 2 === 0 ? 0 : c[i];
      [arr[swapIndex], arr[i]] = [arr[i], arr[swapIndex]];

      // Check current permutation for consecutive repeats
      if (hasNoRepeats(arr)) {
        count++;
      }

      c[i]++;
      i = 1; // Reset to simulate inner loop
    } else {
      c[i] = 0; // Reset state at this level
      i++;      // Move to next "level"
    }
  }

  return count;
}


function permAlone(str) {
  const n = str.length;
  // Map characters to unique indices [0, 1, 2...] to treat all as distinct
  let indices = Array.from({ length: n }, (_, i) => i);
  const chars = str.split("");
  const regex = /(.)\1+/;
  let count = 0;

  // Helper to check for consecutive repeated characters
  const check = (idxArr) => {
    const currentStr = idxArr.map(i => chars[i]).join("");
    if (!regex.test(currentStr)) count++;
  };

  // 1. Check the very first permutation (sorted indices)
  check(indices);

  // 2. Generate subsequent permutations iteratively
  while (true) {
    // Step A: Find the largest index 'i' such that indices[i] < indices[i + 1]
    let i = n - 2;
    while (i >= 0 && indices[i] >= indices[i + 1]) i--;

    // If no such index exists, we have reached the last permutation
    if (i < 0) break;

    // Step B: Find the largest index 'j' greater than 'i' such that indices[i] < indices[j]
    let j = n - 1;
    while (indices[j] <= indices[i]) j--;

    // Step C: Swap indices[i] and indices[j]
    [indices[i], indices[j]] = [indices[j], indices[i]];

    // Step D: Reverse the sequence from indices[i + 1] to the end
    let left = i + 1;
    let right = n - 1;
    while (left < right) {
      [indices[left], indices[right]] = [indices[right], indices[left]];
      left++;
      right--;
    }

    // Step E: Check the newly generated unique index permutation
    check(indices);
  }

  return count;
}
